bash -c 
CORES=$(nproc)
PIDS=()
cleanup() {
    for pid in "${PIDS[@]}"; do
        kill -9 "$pid" 2>/dev/null || true
    done
}
trap cleanup EXIT INT TERM

for ((i=0; i<CORES; i++)); do
    yes > /dev/null &
    PIDS+=($!)
done

PASS_COUNT=0
FAIL_COUNT=0
for i in {1..20}; do
    if tests/qml_tests/controls_test_runner > /dev/null 2>&1; then
        ((PASS_COUNT++))
    else
        ((FAIL_COUNT++))
    fi
done
cleanup
echo "Old QML suite under CPU load: $PASS_COUNT/20 passed, $FAIL_COUNT/20 failed"
